julia> using WaveAcoustics

julia> results = convergence_study_coupled(
           input_data = example1_manufactured(2.4),
           solver = CrankNicolson3(),
           fe = Lagrange{1}(),
           t_end = 1.0,
           Nx_exp_range = 3:6);

julia> print_convergence_table(results)
